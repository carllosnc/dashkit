import { FileUpload } from '../dashkit/FileUpload';

# FileUpload

A drag-and-drop file upload dropzone with file size validation, per-file progress bars, and status indicators.

## Basic usage

<div className="not-prose max-w-lg">
  <FileUpload
    label="Project Assets"
    accept="image/*,.pdf"
    maxSize={5 * 1024 * 1024}
    defaultValue={[
      { id: '1', name: 'architecture.pdf', size: 1245184, progress: 68, status: 'uploading' },
      { id: '2', name: 'brand-mark.svg', size: 24576, progress: 100, status: 'completed' }
    ]}
    description="Supports SVG, PNG, JPG, and PDF up to 5 MB."
  />
</div>

```tsx
import { FileUpload } from '@/dashkit';

export function Example() {
  return (
    <FileUpload
      label="Project Assets"
      accept="image/*,.pdf"
      maxSize={5 * 1024 * 1024}
      onFilesAdded={(files) => console.log(files)}
    />
  );
}
```
