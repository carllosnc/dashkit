import { useState } from 'react';
import { FiFilePlus, FiSearch, FiSettings, FiTerminal } from 'react-icons/fi';
import { CommandPalette, type CommandPaletteItem } from '../dashkit/CommandPalette';
import { Button } from '../dashkit/Button';

export function CommandPaletteDemo() {
  const [open, setOpen] = useState(false);
  const items: CommandPaletteItem[] = [
    {
      id: 'new-doc',
      label: 'Create Document',
      description: 'Start a fresh markdown report',
      group: 'Actions',
      keywords: ['new', 'file'],
      shortcut: ['⌘', 'N'],
      icon: <FiFilePlus />
    },
    {
      id: 'run-cli',
      label: 'Run Diagnostics',
      description: 'Verify Dashkit registry health',
      group: 'Actions',
      shortcut: ['⌘', 'D'],
      icon: <FiTerminal />
    },
    {
      id: 'search-docs',
      label: 'Search Components',
      group: 'Navigation',
      shortcut: ['⌘', 'K'],
      icon: <FiSearch />
    },
    {
      id: 'settings',
      label: 'Workspace Settings',
      group: 'Navigation',
      shortcut: ['⌘', ','],
      icon: <FiSettings />
    }
  ];

  return (
    <div className="not-prose">
      <Button variant="outlined" onClick={() => setOpen(true)}>
        Open Command Palette (⌘K)
      </Button>
      <CommandPalette
        open={open}
        onOpenChange={setOpen}
        items={items}
        footer={<span>Use ↑↓ to navigate, ↵ to select, ESC to close</span>}
      />
    </div>
  );
}

# CommandPalette

A fast, keyboard-first command menu (`Cmd+K` / `Ctrl+K`) for triggering actions, searching records, and navigating workspaces.

## Basic usage

<CommandPaletteDemo />

```tsx
import { CommandPalette } from '@/dashkit';

export function Example() {
  const [open, setOpen] = useState(false);

  return (
    <CommandPalette
      open={open}
      onOpenChange={setOpen}
      items={[
        { id: 'new-doc', label: 'Create Document', group: 'Actions', shortcut: ['⌘', 'N'] },
        { id: 'settings', label: 'Workspace Settings', group: 'Navigation', shortcut: ['⌘', ','] }
      ]}
    />
  );
}
```
