import { TagInput } from '../dashkit/TagInput';

# TagInput

A multi-select tag input supporting keyboard navigation (`ArrowLeft`, `ArrowRight`, `Backspace`, `Delete`), delimiter commits, and autocomplete suggestions.

## Basic usage

<div className="not-prose max-w-md">
  <TagInput
    label="Skills"
    defaultValue={['TypeScript', 'React']}
    suggestions={['Tailwind CSS', 'Node.js', 'GraphQL', 'Vitest', 'Bun']}
    description="Press Enter or comma to add a tag. Use ArrowLeft/Delete to remove."
    placeholder="Add a skill..."
  />
</div>

```tsx
import { TagInput } from '@/dashkit';

export function Example() {
  return (
    <TagInput
      label="Skills"
      defaultValue={['TypeScript', 'React']}
      suggestions={['Tailwind CSS', 'Node.js', 'GraphQL']}
      placeholder="Add a skill..."
    />
  );
}
```
